country_event = { # Focus: Truce (SHO event)
	id = amaterasu.7
	title = amaterasu.7.t
	desc = amaterasu.7.d
	picture = GFX_report_news_peace_dove

	is_triggered_only = yes

	option = {
		name = amaterasu.5.a	# accept
		hidden_effect = { AMO = { country_event = amaterasu.8 } }
		if = {
			limit = { any_owned_state = { is_controlled_by = AMO } }
			custom_effect_tooltip = AMO_resolution_end_war_tt
			every_owned_state = {
				limit = { is_controlled_by = AMO }
				hidden_effect = { AMO = { transfer_state = PREV } }
			}
		}
		if = {
			limit = { AMO = { any_owned_state = { is_controlled_by = SHO } } }
			custom_effect_tooltip = AMO_resolution_end_war_tt2
			AMO = {
				every_owned_state = {
					limit = { is_controlled_by = SHO }
					hidden_effect = { SHO = { transfer_state = PREV } }
				}
			}
		}
		ai_chance = {
			base = 100
			modifier = {
				factor = 0
				strength_ratio = {
					tag = AMO
					ratio > 5
				}
			}
		}
	}
	option = {
		name = amaterasu.5.b	# refuse
		AMO = { country_event = { id = amaterasu.9 hours = 2 } }
		ai_chance = {
			base = 200
			modifier = {
				factor = 0.5
				strength_ratio = {
					tag = AMO
					ratio < 1
				}
			}
			modifier = {
				factor = 0.5
				check_variable = { global.amo_rebellion_score > 33 }
			}
			modifier = {
				factor = 0.5
				any_enemy_country = {
					NOT = { tag = AMO }
				}
			}
			modifier = {
				factor = 0
				strength_ratio = {
					tag = AMO
					ratio < 5
				}
				OR = {
					has_war_with = PRO
					has_war_with = KAN
					has_war_with = TOH
					has_war_with = CHU	# EZO, KYU and SHI are a bit far away
				}
			}
		}
	}
}

country_event = { # A Common Enemy 
	id = amaterasu.31
	title = amaterasu.31.t
	desc = amaterasu.31.d
	picture = GFX_report_event_modern_samurai

	is_triggered_only = yes
	
	major = yes

	option = {
		name = amaterasu.31.b
		if = { limit = { tag = CHU } AMO = { custom_effect_tooltip = AMO_trade_subsidize_spirit_chu_tt_display } }
		if = { limit = { tag = KAN } AMO = { custom_effect_tooltip = AMO_trade_subsidize_spirit_kan_tt_display } }
		if = { limit = { tag = PRO } AMO = { custom_effect_tooltip = AMO_trade_subsidize_spirit_pro_tt_display } }
		if = { limit = { tag = TOH } AMO = { custom_effect_tooltip = AMO_trade_subsidize_spirit_hto_tt_display } }

		hidden_effect = { AMO = { country_event = amaterasu.312 } }
	}
	option = {
		name = amaterasu.31.a
		trigger = { has_equipment = { infantry_equipment > 999 } }
		set_temp_variable = { amount_equip = 1000 }
		if = { limit = { has_equipment = { infantry_equipment > 9999 } } set_temp_variable = { amount_equip = 5000 } }
		else_if = { limit = { has_equipment = { infantry_equipment > 7999 } } set_temp_variable = { amount_equip = 4000 } }
		else_if = { limit = { has_equipment = { infantry_equipment > 5999 } } set_temp_variable = { amount_equip = 3000 } }
		else_if = { limit = { has_equipment = { infantry_equipment > 3999 } } set_temp_variable = { amount_equip = 2000 } }
		effect_tooltip = {
			send_equipment = {
				equipment = infantry_equipment
				amount = var:amount_equip
				target = AMO
			}
		}
		hidden_effect = { AMO = { country_event = amaterasu.311 } }
	}
	option = {
		name = amaterasu.31.c
		trigger = { has_manpower > 2999 }
		set_temp_variable = { amount_volunteers = 3000 }
		if = { limit = { has_manpower > 8999} set_temp_variable = { amount_volunteers = 9000 } }
		else_if = { limit = { has_manpower > 7499} set_temp_variable = { amount_volunteers = 7500 } }
		else_if = { limit = { has_manpower > 5999} set_temp_variable = { amount_volunteers = 6000 } }
		else_if = { limit = { has_manpower > 4499} set_temp_variable = { amount_volunteers = 4500 } }
		effect_tooltip = {
			AMO = { add_manpower = var:amount_volunteers }
			multiply_temp_variable = { amount_volunteers = -1 }
			add_manpower = var:amount_volunteers
		}
		hidden_effect = { AMO = { country_event = amaterasu.313 } }
	}
	option = {
		name = amaterasu.31.no
		hidden_effect = { AMO = { country_event = amaterasu.314 } }
	}
}
