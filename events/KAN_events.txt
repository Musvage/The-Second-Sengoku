country_event = { # Help the Knights (AMO event)
	id = kansai.11
	title = kansai.11.t
	desc = kansai.11.desc
	picture = GFX_report_event_ger_civil_war
	is_triggered_only = yes
	
	option = {	# accept 
		name = kansai.11.a
		### SINCE THIS ONLY HAPPENS EARLY GAME, WE CAN ASSUME THE BORDERS ARE THE SAME AS IN 1.1.1936
		# select spawn state
		if = {
			limit = { AMO = { controls_state = 118 } }
			set_global_flag = KAN_help_knights_state_118
		} else_if = {
			limit = { AMO = { controls_state = 63 } }
			set_global_flag = KAN_help_knights_state_63
		} else_if = {
			limit = { AMO = { controls_state = 44 } }
			set_global_flag = KAN_help_knights_state_44
		} else = {
			set_global_flag = KAN_help_knights_no_state
		}
		if = {
			limit = { has_global_flag = KAN_help_knights_no_state }
			# spawn units directly in Notojima
			AMO = { load_oob = AMO_spawn_196 }
		} else = {
			# spawn unit via switching province controller and loading specific oob
			custom_effect_tooltip = kansai.11.tt3
			if = {
				limit = { has_global_flag = KAN_help_knights_state_118 }
				AMO = {
					set_province_controller = 698
					hidden_effect = { load_oob = AMO_spawn_118 }
				}
			} else_if = {
				limit = { has_global_flag = KAN_help_knights_state_44 }
				AMO = {
					set_province_controller = 5744
					hidden_effect = { load_oob = AMO_spawn_44 }
				}
			} else_if = {
				limit = { has_global_flag = KAN_help_knights_state_63 }
				AMO = {
					set_province_controller = 4564
					hidden_effect = { 
						set_province_controller = 3911 
						load_oob = AMO_spawn_63
					}
				}
			}
		}
		custom_effect_tooltip = KAN_disonorable_action_KAN
		custom_effect_tooltip = loc_paragraph
		hidden_effect = {
			add_ai_strategy = { type = befriend id = KAN value = 50 }
			KAN = {
				add_manpower = -9000
				add_equipment_to_stockpile = {
					type = infantry_equipment
					amount = -900
				}
				add_ai_strategy = { type = befriend id = AMO value = 50 }
			}
		}
		give_military_access = KAN
		KAN = { give_military_access = AMO }
		set_country_flag = accepted_help_from_KAN
		hidden_effect = { country_event = { id = kansai.12 hours = 6 } }
		ai_chance = {
			base = 4
		}
	}
	option = {	# refuse
		name = kansai.11.b
		hidden_effect = { KAN = { country_event = { id = kansai.13 hours = 6 } } }
		ai_chance = {
			base = 1
			modifier = {
				factor = 16
				OR = {
					has_wargoal_against = KAN
					KAN = { has_wargoal_against = AMO }
				}
			}
		}
	}
}
