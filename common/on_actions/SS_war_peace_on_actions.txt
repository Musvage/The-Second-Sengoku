on_actions = {
	on_declare_war = {	# FROM is war target. # ROOT is for the country who is declaring war
		effect = {
			if = {
				limit = {
					ROOT = { tag = KAN }
					FROM = { has_country_flag = accepted_help_from_KAN }
				}
				FROM = {
					country_event = { id = kansai.51 hours = 6 random = 18 }
					clr_country_flag = accepted_help_from_KAN
				}
			}
			if = {
				limit = {
					FROM = { tag = KAN }
					ROOT = { has_country_flag = accepted_help_from_KAN }
				}
				ROOT = {
					country_event = { id = kansai.52 hours = 6 random = 18 }
					clr_country_flag = accepted_help_from_KAN
				}
			}
		}
	}
	on_generate_wargoal = {	# ROOT is the wargoal owner # FROM is the wargoal target
		effect = {
			if = {
				limit = {
					ROOT = { tag = KAN }
					FROM = { has_country_flag = accepted_help_from_KAN }
				}
				FROM = {
					country_event = { id = kansai.51 hours = 6 random = 18 }
					clr_country_flag = accepted_help_from_KAN
				}
			}
			if = {
				limit = {
					FROM = { tag = KAN }
					ROOT = { has_country_flag = accepted_help_from_KAN }
				}
				ROOT = {
					country_event = { id = kansai.52 hours = 6 random = 18 }
					clr_country_flag = accepted_help_from_KAN
				}
			}
		}
	}
	on_war_relation_added = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = { ROOT = { tag = IMS } FROM = { tag = CHU } }
						AND = { FROM = { tag = IMS } ROOT = { tag = CHU } }
					}
					IMS = {
						has_relation_modifier = {
							target = CHU
							modifier = generic_cooperative_posture
						}
					}
				}
				IMS = {
					remove_opinion_modifier = {
						target = CHU
						modifier = generic_cooperative_posture
					}
				}
			}
			if = {
				limit = {
					OR = {
						AND = { ROOT = { tag = IMS } FROM = { tag = SHI } }
						AND = { FROM = { tag = IMS } ROOT = { tag = SHI } }
					}
					IMS = {
						has_relation_modifier = {
							target = SHI
							modifier = generic_cooperative_posture
						}
					}
				}
				IMS = {
					remove_opinion_modifier = {
						target = SHI
						modifier = generic_cooperative_posture
					}
				}
			}
		}
	}
	on_peace = {
		effect = {
			if = {
				limit = {
					tag = AMO
					has_global_flag = amo_sho_scripted_war_ongoing
					AMO = { NOT = { has_war_with = SHO } }
				}
				clr_global_flag = amo_sho_scripted_war_ongoing
			}
		}
		effect = {
			### PLANNED ECONOMY ###
			if = {
				limit = { has_government = communism }
				planned_economy_update = yes
			}
		}
	}
	
	on_release_as_puppet = {	# ROOT is the nation being released, FROM is the overlord.
		effect = {
			if = {
				limit = { NOT = { check_variable = { FROM.country_religion = ROOT.country_religion } } }
				if = {
					limit = { FROM = { has_religion_shinto = yes } }
					ROOT = { set_religion_shinto = yes }
				} else_if = {
					limit = { FROM = { has_religion_christianity = yes } }
					ROOT = { set_religion_christianity = yes }
				} else_if = {
					limit = { FROM = { has_religion_taoism = yes } }
					ROOT = { set_religion_taoism = yes }
				} else_if = {
					limit = { FROM = { has_religion_confucianism = yes } }
					ROOT = { set_religion_confucianism = yes }
				} else_if = {
					limit = { FROM = { has_religion_secularism = yes } }
					ROOT = { set_religion_secularism = yes }
				} else_if = {
					limit = { FROM = { has_religion_cthulhu = yes } }
					ROOT = { set_religion_cthulhu = yes }
				} else = {
					ROOT = { set_religion_buddhism = yes }	# default, if there is any error, Buddhism may not raise eyebrows
				}
			}
		}
	}
	
	on_capitulation_immediate = {
		effect = {
			if = {
				limit = { ROOT = { tag = IMS } }
				clr_global_flag = seto_crisis_active
			}
		}
		effect = {
			if = {
				limit = { ROOT = { tag = AMO has_war_with = SHO has_active_mission = AMO_ongoing_war_amo AMO = { has_capitulated = no } } }
				SHO = { country_event = amaterasu.5	}	# Shogun Total Victory (triggered by on_actions)
			}
			if = {
				limit = { ROOT = { tag = SHO has_war_with = AMO has_active_mission = AMO_ongoing_war_sho SHO = { has_capitulated = no } } }
				AMO = { country_event = amaterasu.10 }	# Knights Total Victory (triggered by on_actions)
			}
		}
	}
	on_capitulation = { # ROOT is capitulated country, FROM is winner.
		effect = {
			# scripted white peaces when Japanese are fighting Asian mainlanders without allies in Japan
			if = {
				limit = {
					ROOT = {	# capitulated is japanese
						capital_scope = {
							NOT = {
								is_on_continent = korea
								is_on_continent = asia
								is_on_continent = sakhalin
							}
						}
						is_in_faction = yes
						is_faction_leader = no
						all_allied_country = {	# no more Japanese active in the faction
							OR = {
								is_faction_leader = yes
								has_capitulated = yes
								NOT = { capital_scope = { is_on_continent = korea is_on_continent = asia is_on_continent = sakhalin } }
							}
						}
						faction_leader = {
							has_capitulated = no
							capital_scope = { OR = { is_on_continent = korea is_on_continent = asia is_on_continent = sakhalin } }	# faction leader is not japanese
						}
					}
					FROM = {	# winner is japanese
						capital_scope = {
							NOT = {
								is_on_continent = korea
								is_on_continent = asia
								is_on_continent = sakhalin
							}
						}
						OR = {	# can negotiate its own peace
							is_in_faction = no
							is_faction_leader = yes
						}
					}
				}
				ROOT = { faction_leader = { FROM = { country_event = { id = scriptedpeace.1 hours = 12 } } } }
			}
		}
	}
	on_capitulation_immediate = {
		effect = {
			log = "on_capitulation_immediate [ROOT.GetName] is the loser, [FROM.GetName] is the winner."
		}
		effect = {
			if = {
				limit = {
					has_global_flag = amo_sho_scripted_war_ongoing
				}
				if = {
					limit = { ROOT = { tag = SHO } FROM = { tag = AMO } }	# AMO wins
					AMO = { country_event = amaterasu.10 remove_decision = AMO_ongoing_war_amo }
					SHO = { remove_decision = AMO_ongoing_war_sho }
					clr_global_flag = amo_sho_scripted_war_ongoing
				} else_if = {
					limit = { ROOT = { tag = AMO } FROM = { tag = SHO } }	# SHO wins
					SHO = { country_event = amaterasu.5 remove_decision = AMO_ongoing_war_sho }
					AMO = { remove_decision = AMO_ongoing_war_amo }
					clr_global_flag = amo_sho_scripted_war_ongoing
				}
			}
		}
	}
	on_peaceconference_started = {
		effect = {
			log = "on_peaceconference_started [ROOT.GetName] is the winner, [FROM.GetName] is the loser."
		}
	}
	on_peaceconference_ended = {
		effect = {
			log = "on_peaceconference_ended [ROOT.GetName] is the winner, [FROM.GetName] is the loser."
		}
	}
}
